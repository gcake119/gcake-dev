// One-shot case: append to a disposable copy of cloud-sync.acceptance.test.ts in the same directory;
// run node --import tsx --test --test-name-pattern='One-shot equal-input' <copy>, then remove only that copy.
test('One-shot equal-input scheduling comparison',async()=>{
 const {execFileSync}=await import('node:child_process');const {transpileModule,ModuleKind}=await import('typescript');const {writeFile}=await import('node:fs/promises');
 const baseline='13f71b8584b8307e0f4bfd3bf84f3bde500dd11e';
 const original=execFileSync('git',['show',baseline+':apps/search-worker/src/sync.ts'],{encoding:'utf8'});
 const dependencies=Object.fromEntries(await Promise.all(['admission','deployed','service','runtime','provider'].map(async n=>['./'+n,await import('../../apps/search-worker/src/'+n+'.ts')])));
 const exports:any={};new Function('require','exports',transpileModule(original,{compilerOptions:{module:ModuleKind.CommonJS}}).outputText)((id:string)=>dependencies[id],exports);
 const results=[];
 for(const delay of [1,2])for(const variant of ['original','candidate']){
  const h=await setup(33);try{
   let clock=Date.now(),rounds=0,reads=0,describes=0,d1=0;const start=clock,visible=new Map<string,number>();
   const upsert=h.env.VECTOR_INDEX.upsert,get=h.env.VECTOR_INDEX.getByIds,describe=h.env.VECTOR_INDEX.describe,prepare=h.env.DISCOVERY_DB.prepare;
   h.env.DISCOVERY_DB.prepare=(q:string)=>{const stmt=prepare(q);for(const key of ['first','all','run'] as const){const method=stmt[key].bind(stmt);(stmt as any)[key]=async(...args:any[])=>{d1++;return method(...args);};}return stmt;};
   h.env.VECTOR_INDEX.upsert=async v=>{for(const x of v)visible.set(x.id,clock+delay*60000);return upsert(v);};
   h.env.VECTOR_INDEX.getByIds=async ids=>{reads++;return(await get(ids)).filter(v=>clock>=(visible.get(v.id)??0));};
   h.env.VECTOR_INDEX.describe=async()=>{describes++;return describe();};
   let status='';while(status!=='promoted'&&rounds<20){status=(await (variant==='original'?exports.synchronizeIndex(h.env,{read:async()=>({corpus:h.c,corpusId:h.c.corpusId}),now:()=>clock}):h.run({now:()=>clock}))).status;rounds++;if(status!=='promoted')clock+=60000;}
   assert.equal(status,'promoted');const reservations=h.sql.prepare('SELECT sum(tokens) tokens,sum(units) units,sum(worker) worker,sum(admission) admission,sum(storage) storage FROM discovery_reservations').get()!;
   assert.ok(Number(reservations.admission)>=d1,`${variant} insufficient D1 budget`);
   results.push({variant,passages:33,batchSize:16,visibilityDelayMinutes:delay,rounds,modeledElapsedMs:clock-start,embeddings:h.calls()[0],uploads:h.calls()[1],reads,describes,d1Statements:d1,reservations});
  }finally{h.close();}
 }
 await writeFile('openspec/changes/fix-discovery-corpus-identity/evidence/equal-input-comparison.json',JSON.stringify({baseline,scope:'offline controlled-clock, one deterministic sample per delay and variant; no provider or billing measurement',results},null,2)+'\n');
});
